      return {
        totalConcerns,
        concernsByStatus,
        concernsByCategory,
        concernsBySeverity,
        resolutionRate,
        averageResolutionTime,
        total,
        toBeDone,
        addressed,
        rejected,
        byCategory,
        bySeverity
      };

    } catch (error) {
      console.error('Error calculating concern statistics:', error);
      throw error instanceof Error ? error : new Error('Unknown error calculating statistics');
    }
  }
